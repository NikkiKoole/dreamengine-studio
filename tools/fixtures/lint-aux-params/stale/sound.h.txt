typedef struct Instrument {
  float eng_p[4];
} Instrument;

typedef struct Voice {
  float eng_p[3];        // EXPECT: decl-disagree (4 vs 3)
} Voice;

// Only ONE bound left — the second was refactored into something this regex can't see.
// EXPECT: bound-count. Silence here is the failure mode: with fewer than two bounds found,
// the "every bound equals the width" loop passes vacuously.
void instrument_mode(int slot, int idx, float v) {
  if (idx < 0 || idx >= 4) return;
  queue_request(SR_ENG_TUNE, slot, idx, v);
}

static void voice_note_on(Voice *v, Instrument *ins) {
  v->eng_p[0] = ins->eng_p[0];
  v->eng_p[1] = ins->eng_p[1];
  v->eng_p[2] = ins->eng_p[2];
  v->eng_p[3] = ins->eng_p[3];
}
